async function loadAdminDashboard() { try { const tbody = document.getElementById('adminClinicsTable'); tbody.innerHTML = 'جاري جلب أصحاب العيادات (owner)...'; // جلب كافة المستخدمين من مجموعة sys_users const snapshot = await db.collection("sys_users").get(); if (snapshot.empty) { tbody.innerHTML = 'لا توجد حسابات مسجلة في sys_users.'; return; } tbody.innerHTML = ''; const now = new Date(); let foundOwnersCount = 0; snapshot.forEach(doc => { const data = doc.data(); const docId = doc.id; // استثناء حساب السوبر أدمن الخاص بك if (data.username === 'abumousa2027') return; // الشرط الأساسي: إظهار فقط من هو صاحب عيادة (role === 'owner') if (data.role === 'owner') { foundOwnersCount++; const facilityName = data.facility_name || data.full_name || data.username || docId; const planTitle = data.subscription_plan || 'غير محدد'; const status = data.subscription_status || 'active'; const expiryISO = data.expiry_date; let expiryDisplay = 'لم يحدد تاريخ بعد'; let isExpired = false; if (expiryISO) { const expDate = new Date(expiryISO); expiryDisplay = expDate.toLocaleString('ar-EG', { dateStyle: 'medium', timeStyle: 'short' }); if (now > expDate) { isExpired = true; } } tbody.innerHTML += ` ${facilityName}
يوزر: ${data.username || docId} ${planTitle} ${expiryDisplay} ${status === 'blocked' ? 'موقوف يدويًا 🛑' : (isExpired ? 'منتهي الصلاحية ❌' : 'ساري المفعول ✔')} `; } }); if (foundOwnersCount === 0) { tbody.innerHTML = 'لا يوجد أي مستخدم بصلاحية (owner) حتى الآن. تأكد من ضبط حقل role ليكون owner في فايرستور.'; } } catch (err) { console.error("Error loading owners:", err); alert("حدث خطأ أثناء جلب أصحاب العيادات."); } }
