async function loadAdminDashboard() { try { const tbody = document.getElementById('adminClinicsTable'); tbody.innerHTML = 'جاري جلب أصحاب العيادات من sys_users...'; // 1. جلب كافة المستندات من sys_users const snapshot = await db.collection("sys_users").get(); if (snapshot.empty) { tbody.innerHTML = 'لا يوجد مستخدمين مسجلين في sys_users.'; return; } tbody.innerHTML = ''; const now = new Date(); snapshot.forEach(doc => { const data = doc.data(); const docId = doc.id; // استثناء حسابك الشخصي كـ Super Admin من القائمة (abumousa2027) if (data.username === 'abumousa2027') return; // تصفية أصحاب العيادات فقط (الذين يملكون صلاحية admin أو facility_owner) // إذا لم تكن كلمة role مستخدمة عندك، سيتم عرض جميع المستخدمين لتختار منهم const facilityName = data.facility_name || data.full_name || data.username || docId; const userRole = data.role || 'غير محدد'; const planTitle = data.subscription_plan || 'غير محدد'; const status = data.subscription_status || 'active'; const expiryISO = data.expiry_date; let expiryDisplay = 'لم يحدد تاريخ بعد'; let isExpired = false; if (expiryISO) { const expDate = new Date(expiryISO); expiryDisplay = expDate.toLocaleString('ar-EG', { dateStyle: 'medium', timeStyle: 'short' }); if (now > expDate) { isExpired = true; } } tbody.innerHTML += ` ${facilityName}
يوزر: ${data.username || docId} | الصفة: ${userRole} ${planTitle} ${expiryDisplay} ${status === 'blocked' ? 'موقوف يدويًا 🛑' : (isExpired ? 'منتهي الصلاحية ❌' : 'ساري المفعول ✔')} `; }); } catch (err) { console.error("Error loading clinics from sys_users:", err); alert("حدث خطأ أثناء جلب حسابات العيادات من sys_users."); } }
